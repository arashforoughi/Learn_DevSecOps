## 34.2 Phase 1: Foundation Deployment (8-10 hours)

### Step 1.1: Host Preparation

```bash
#!/bin/bash
# 01-host-preparation.sh

# 1. Update system
sudo apt-get update && sudo apt-get upgrade -y

# 2. Install hypervisor
sudo apt-get install -y \
  qemu-kvm \
  libvirt-daemon-system \
  libvirt-clients \
  virt-manager \
  terraform \
  ansible

# 3. Enable nested virtualization
cat <<EOF | sudo tee /etc/modprobe.d/kvm-nested.conf
options kvm_intel nested=1
options kvm_intel enable_shadow_vmcs=1
options kvm_intel enable_apicv=1
options kvm_intel ept=1
EOF

# 4. Create lab network
sudo virsh net-define /dev/stdin << 'NETXML'
<network>
  <name>virbr-lab</name>
  <forward mode='nat'/>
  <bridge name='virbr-lab' stp='on' delay='0'/>
  <ip address='10.0.2.1' netmask='255.255.255.0'>
    <dhcp>
      <range start='10.0.2.100' end='10.0.2.254'/>
    </dhcp>
  </ip>
</network>
NETXML

sudo virsh net-start virbr-lab
sudo virsh net-autostart virbr-lab

# 5. Create base image
sudo virt-install \
  --name ubuntu-base \
  --memory 2048 \
  --vcpus 2 \
  --disk size=20 \
  --location http://archive.ubuntu.com/ubuntu/dists/jammy/main/installer-amd64/ \
  --os-type linux \
  --os-variant ubuntu22.04

echo "Host preparation complete!"
```

### Step 1.2: Deploy Foundation VMs (DNS, Identity, Edge)

```bash
#!/bin/bash
# 01-deploy-foundation.sh

# Use Terraform to deploy foundation VMs
cd ~/lab/opentofu/devsecops-lab

# Initialize Terraform
terraform init

# Deploy only Phase 1 VMs
terraform apply \
  -target='libvirt_domain.foundation' \
  -var="deploy_phase=1"

# Wait for VMs to boot
sleep 60

# Run Ansible configuration
cd ~/lab/ansible/devsecops-lab
ansible-playbook -i inventory.ini playbooks/phase-1.yml

# Verify deployment
echo "Verifying Phase 1 deployment..."
ansible all -m ping
```

### Step 1.3: Validation

```bash
#!/bin/bash
# 01-validate-phase1.sh

echo "=== PHASE 1 VALIDATION ==="

# Check DNS
dig @10.0.2.10 devsecops.com +short
dig @10.0.2.10 sec-idm-01.devsecops.com +short

# Check Identity
curl -k https://auth.devsecops.com/health

# Check Edge routing
curl -k https://10.0.2.12/health

# Test HTTPS
openssl s_client -connect 10.0.2.12:443 -servername devsecops.com </dev/null

echo "Phase 1 validation complete!"
```

## 34.3 Phase 2: Automation Deployment (6-8 hours)

```bash
#!/bin/bash
# 02-deploy-automation.sh

cd ~/lab/opentofu/devsecops-lab

# Build Packer golden image
cd ~/lab/packer
packer build -var="qemu_binary=/usr/bin/qemu-system-x86_64" ubuntu-22.04/ubuntu-22.04.pkr.hcl

# Deploy Phase 2 VMs with Terraform
cd ~/lab/opentofu/devsecops-lab

cd ~/lab/ansible/devsecops-lab

# Verify
echo "Verifying Phase 2..."
ansible-playbook playbooks/validate/phase-2.yml
```

## 34.4 Rapid Deployment Option

### Deploy Everything in 30 Minutes

```bash
#!/bin/bash
# rapid-deploy.sh - Deploy complete platform in 30 min

set -e

echo "🚀 Starting rapid DevSecOps deployment..."

# Clone all repositories
git clone https://github.com/devsecops/lab ~/lab

cd ~/lab

# 1. Generate Terraform variables
cat > terraform.tfvars << 'VARS'
deployment_name = "devsecops-lab"
deploy_all_phases = true
qemu_machine = "q35"
network_cidr = "10.0.2.0/24"
VARS

# 2. Initialize and deploy with Terraform
cd opentofu/devsecops-lab
terraform init
terraform plan -out=tfplan
terraform apply tfplan

# 3. Get VM IPs
echo "Waiting for VMs to boot..."
sleep 120

# 4. Configure with Ansible
cd ../../ansible/devsecops-lab
ansible-playbook playbooks/site.yml -i inventory.ini

# 5. Validate
echo "Running validation..."
ansible-playbook playbooks/validate/all.yml

echo "✅ Deployment complete!"
kubectl get nodes
```

## 34.5 Customization Guide

### Adapt to Your Environment

```bash
# 1. Change network CIDR
sed -i 's/10.0.2.0\/24/YOUR_CIDR/g' opentofu/variables.tf

# 2. Change domain
sed -i 's/devsecops.com/yourdomain.com/g' ansible/roles/dns/templates/zones.j2

# 3. Scale worker nodes
# Edit opentofu/vms.tf, change:
#   worker_count = 2 → worker_count = 5

# 4. Adjust resources
# Edit opentofu/vms.tf for CPU, RAM, disk sizes

# 5. Use external registry
sed -i 's/harbor.devsecops.com/your-registry.com/g' kubernetes/manifests/*.yaml
```

## 34.6 Troubleshooting Deployment

```bash
# VMs not starting
virsh list --all
virsh start vm-name

# Network issues
virsh net-list
virsh net-dumpxml virbr-lab

# DNS not resolving
nslookup devsecops.com 10.0.2.10

# Kubernetes not ready
kubectl get nodes -o wide
kubectl describe node cp-01

# Terraform state issues
terraform state list
terraform state show libvirt_domain.example
terraform state rm resource_type.name  # Remove and reapply

# Ansible inventory issues
ansible all -i inventory.ini -m ping
ansible-inventory -i inventory.ini --list
```

## 34.7 Migration Path

### From Existing Infrastructure

```bash
# 1. Inventory your existing systems
ansible-inventory -i existing-inventory.ini --list > systems.json

# 2. Map to DevSecOps architecture
# Review: existing-systems → devsecops-vms.txt

# 3. Configure migration
terraform import libvirt_domain.existing existing-vm-name

# 4. Update DNS
# Point devsecops.com domain to new infrastructure

# 5. Gradual cutover
# Run parallel for 2 weeks
# Monitor both systems
# Cutover traffic after validation

# 6. Decommission old systems
terraform destroy -target='existing_resource'
```

## 34.8 Disaster Recovery Setup

```bash
#!/bin/bash
# Setup backup and recovery

# 1. Backup configuration
mkdir -p /mnt/backups/{configs,vms,databases}

# 2. Export VM definitions
virsh dumpxml cp-01 > /mnt/backups/vms/cp-01.xml

# 3. Backup etcd
ETCDCTL_API=3 etcdctl snapshot save /mnt/backups/databases/etcd.db

# 4. Backup Vault
vault write sys/storage/raft/snapshot file=@/mnt/backups/databases/vault-snapshot

# 5. Test recovery
virsh create /mnt/backups/vms/cp-01.xml
```
