## 17.1 Complete Pipeline Flow

```
Developer commits code
    ↓
GitLab trigger CI/CD pipeline
    ├─ Stage 1: BUILD
    │   └─ Compile, install dependencies
    ├─ Stage 2: TEST
    │   └─ Unit tests, integration tests
    ├─ Stage 3: SCAN (Security)
    │   ├─ SonarQube SAST scan
    │   └─ Dependency scan
    ├─ Stage 4: QUALITY GATE
    │   └─ If failed: STOP pipeline
    ├─ Stage 5: CODE REVIEW
    │   └─ Merge request approval needed
    ├─ Stage 6: BUILD IMAGE
    │   └─ Create Docker container image
    ├─ Stage 7: PUSH REGISTRY
    │   └─ Push image to Harbor
    ├─ Stage 8: SCAN IMAGE
    │   └─ Harbor/Trivy vulnerability scan
    ├─ Stage 9: COMPLIANCE CHECK
    │   ├─ DefectDojo findings
    │   ├─ SLA check
    │   └─ Critical vulnerability check
    ├─ Stage 10: IMAGE APPROVAL
    │   └─ Manual security team approval
    └─ Stage 11: READY FOR DEPLOYMENT
        └─ Image signed and approved

[Image ready for deployment to staging/production]
```

## 17.2 Complete .gitlab-ci.yml

Here's the complete GitLab CI/CD configuration that orchestrates everything:

```yaml
stages:
  - build
  - test
  - scan_code
  - quality_gate
  - build_image
  - scan_image
  - defectdojo
  - approval
  - deploy

variables:
  REGISTRY_URL: harbor.devsecops.com
  IMAGE_NAME: $REGISTRY_URL/myapp/myapp
  SONAR_HOST_URL: https://sonarqube.devsecops.com
  DEFECTDOJO_URL: https://defectdojo.devsecops.com/api/v2
  VAULT_ADDR: https://vault.devsecops.com
  DOCKER_HOST: tcp://docker:2375
  DOCKER_TLS_VERIFY: ""

#
# ────────────────────────────────────
# STAGE 1: BUILD
# ────────────────────────────────────
#

build:
  stage: build
  image: python:3.11
  script:
    - echo "Building application..."
    - pip install -r requirements.txt
    - python -m py_compile app.py tests/*.py
  artifacts:
    paths:
      - app.py
      - requirements.txt
    expire_in: 1 hour

#
# ────────────────────────────────────
# STAGE 2: TEST
# ────────────────────────────────────
#

test:
  stage: test
  image: python:3.11
  script:
    - pip install -r requirements.txt
    - pip install pytest pytest-cov
    - pytest tests/ -v --cov=. --cov-report=xml --cov-report=term
  coverage: '/TOTAL.*\s+(\d+%)$/'
  artifacts:
    reports:
      coverage_report:
        coverage_format: cobertura
        path: coverage.xml
    paths:
      - coverage.xml
  only:
    - merge_requests

#
# ────────────────────────────────────
# STAGE 3: SCAN CODE (SAST)
# ────────────────────────────────────
#

sonarqube_scan:
  stage: scan_code
  image: sonarsource/sonar-scanner-cli:latest
  script:
    - echo "Scanning code with SonarQube..."
    - sonar-scanner
        -Dsonar.projectKey=myapp
        -Dsonar.sources=.
        -Dsonar.host.url=$SONAR_HOST_URL
        -Dsonar.login=$SONARQUBE_TOKEN
        -Dsonar.python.coverage.reportPath=coverage.xml
  allow_failure: false
  only:
    - merge_requests
    - main

#
# ────────────────────────────────────
# STAGE 4: QUALITY GATE
# ────────────────────────────────────
#

quality_gate:
  stage: quality_gate
  image: curlimages/curl:latest
  script:
    - |
      echo "Checking SonarQube quality gate..."
      QUALITYGATE=$(curl -s \
        -H "Authorization: Bearer $SONARQUBE_TOKEN" \
        "$SONAR_HOST_URL/api/qualitygates/project_status?projectKey=myapp" | \
        jq -r '.projectStatus.status')
      
      if [ "$QUALITYGATE" != "OK" ]; then
        echo "❌ Quality gate FAILED"
        exit 1
      fi
      echo "✅ Quality gate PASSED"
  only:
    - merge_requests
    - main

#
# ────────────────────────────────────
# CODE REVIEW (Manual)
# ────────────────────────────────────
#

# Note: Code review happens in merge request UI
# Cannot merge without approval

#
# ────────────────────────────────────
# STAGE 6: BUILD DOCKER IMAGE
# ────────────────────────────────────
#

build_image:
  stage: build_image
  image: docker:20.10
  services:
    - docker:20.10-dind
  script:
    - echo "Building Docker image..."
    - docker build -t $IMAGE_NAME:$CI_COMMIT_SHA .
    - docker build -t $IMAGE_NAME:latest .
    - echo "✅ Docker image built: $IMAGE_NAME:$CI_COMMIT_SHA"
  only:
    - main
    - master

#
# ────────────────────────────────────
# STAGE 7: PUSH TO HARBOR REGISTRY
# ────────────────────────────────────
#

push_image:
  stage: build_image
  image: docker:20.10
  services:
    - docker:20.10-dind
  script:
    - echo "Pushing image to Harbor..."
    - docker login -u $HARBOR_USERNAME -p $HARBOR_PASSWORD $REGISTRY_URL
    - docker build -t $IMAGE_NAME:$CI_COMMIT_SHA .
    - docker push $IMAGE_NAME:$CI_COMMIT_SHA
    - docker push $IMAGE_NAME:latest
    - echo "✅ Image pushed to Harbor"
  only:
    - main
    - master

#
# ────────────────────────────────────
# STAGE 8: SCAN IMAGE (Container Scanning)
# ────────────────────────────────────
#

scan_image:
  stage: scan_image
  image: curlimages/curl:latest
  script:
    - |
      echo "Waiting for Harbor to scan image..."
      sleep 10  # Wait for scan to complete
      
      VULNERABILITIES=$(curl -s \
        -H "Authorization: Bearer $HARBOR_TOKEN" \
        "https://harbor.devsecops.com/api/v2/projects/myapp/repositories/myapp/artifacts/$CI_COMMIT_SHA/scan" | \
        jq '.scan_overview.critical')
      
      if [ "$VULNERABILITIES" -gt 0 ]; then
        echo "❌ CRITICAL vulnerabilities found in image: $VULNERABILITIES"
        exit 1
      fi
      echo "✅ Image scan passed (no critical vulnerabilities)"
  only:
    - main
    - master

#
# ────────────────────────────────────
# STAGE 9: COMPLIANCE CHECK (DefectDojo)
# ────────────────────────────────────
#

defectdojo_import:
  stage: defectdojo
  image: curlimages/curl:latest
  script:
    - |
      echo "Importing findings to DefectDojo..."
      
      # Get SonarQube findings
      curl -s -X POST \
        -H "Authorization: Token $DEFECTDOJO_API_TOKEN" \
        -H "Content-Type: application/json" \
        -d '{
          "product": "MyApp",
          "scan_type": "SonarQube",
          "scanner_name": "SonarQube",
          "file": "'$CI_PROJECT_DIR'/sonar-report.json"
        }' \
        $DEFECTDOJO_URL/findings/
      
      echo "✅ Findings imported to DefectDojo"
  only:
    - main
    - master

#
# ────────────────────────────────────
# STAGE 10: MANUAL APPROVAL
# ────────────────────────────────────
#

approval:
  stage: approval
  script:
    - echo "⏳ Waiting for security team approval..."
  when: manual  # Manual approval gate
  only:
    - main
    - master

#
# ────────────────────────────────────
# STAGE 11: DEPLOY (Example)
# ────────────────────────────────────
#

deploy_staging:
  stage: deploy
  image: curlimages/curl:latest
  script:
    - |
      echo "Deploying to staging..."
      # This would trigger actual deployment
      # (Chapter 5/Kubernetes shows this)
      
      # For now, just log the deployment
      curl -X POST \
        -H "Authorization: Bearer $DEPLOYMENT_TOKEN" \
        -d "image=$IMAGE_NAME:$CI_COMMIT_SHA" \
        https://deployment.devsecops.com/deploy/staging
      
      echo "✅ Deployed to staging"
  when: manual  # Manual deployment approval
  only:
    - main
    - master

```

## 17.3 Secrets Management in Pipeline

Fetch secrets from Vault during pipeline execution:

```yaml
# Add to pipeline for secure secret retrieval
before_script:
  # Fetch database password from Vault
  - |
    export DB_PASSWORD=$(curl -s \
      -H "X-Vault-Token: $VAULT_TOKEN" \
      $VAULT_ADDR/v1/secret/data/database/mysql | \
      jq -r '.data.data.password')
  
  # Fetch Harbor credentials from Vault
  - |
    export HARBOR_CREDENTIALS=$(curl -s \
      -H "X-Vault-Token: $VAULT_TOKEN" \
      $VAULT_ADDR/v1/secret/data/registry/harbor | \
      jq -r '.data.data')
```

## 17.5 Notifications & Alerting

### Slack Integration

```yaml
# GitLab → Slack notifications
after_script:
  - |
    if [ "$CI_JOB_STATUS" = "success" ]; then
      STATUS="✅ PASSED"
```

## 17.6 Advanced Workflows

### Canary Deployment

```yaml
deploy_canary:
  stage: deploy
  script:
    - |
      # Deploy to 10% of traffic
      # Monitor for errors for 15 minutes
      # If stable, increase to 50%, then 100%
      
      kubectl set image deployment/myapp-canary \
        myapp=$IMAGE_NAME:$CI_COMMIT_SHA \
        --record
```

### Automatic Rollback

```yaml
rollback:
  stage: deploy
  when: manual
  script:
    - |
      # Rollback to previous stable version
      kubectl rollout undo deployment/myapp
      echo "Rolled back to previous version"
```
