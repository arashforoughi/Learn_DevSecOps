```
┌─────────────────────────────────────────────────────────────┐
│              NETWORK SECURITY LAYERS                        │
├─────────────────────────────────────────────────────────────┤
│                                                             │

│  ├─ Entry/exit traffic inspection                         │
│  ├─ Stateful packet filtering                             │

│  ├─ DDoS protection (upstream)                            │
│  └─ Tools: Cisco ASA, Fortinet FortiGate, Palo Alto       │
│                                                             │
│  LAYER 2: HOST FIREWALL (UFW, iptables)                   │

│  LAYER 3: NETWORK SEGMENTATION (VLANs)                    │
│  ├─ Logical network isolation                             │

│  └─ Reduces blast radius of breaches                      │

│  LAYER 4: IDS/IPS (Intrusion Detection)                   │

│  ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━  │
```

## 30.2 UFW Host Firewall Configuration

### Comprehensive UFW Setup

```bash
#!/bin/bash
# firewall-setup.sh - Configure UFW host firewall

# Enable UFW
ufw enable

# Default policies
ufw default deny incoming      # Deny by default (whitelist approach)
ufw default allow outgoing     # Allow outgoing (can be restricted)
ufw default deny routed        # Deny forwarded traffic

# Essential services
ufw allow 22/tcp              # SSH
ufw allow 22/udp              # SSH

# Kubernetes API
ufw allow 6443/tcp            # Control plane API

# Load balancer
ufw allow 80/tcp              # HTTP
ufw allow 443/tcp             # HTTPS

# Application ports (example)
ufw allow from 10.0.0.0/8 to any port 5432  # PostgreSQL from trusted network

# Rate limiting (SSH brute force protection)
ufw limit 22/tcp              # Limit SSH to 6 connections per 30 seconds

# DNS (if running local resolver)
ufw allow 53/tcp              # DNS TCP
ufw allow 53/udp              # DNS UDP

# NTP (time synchronization)
ufw allow 123/udp             # NTP

# ICMP (ping) - allow but limit
ufw allow icmp                 # Allow all ICMP

# View all rules
ufw show added

# Verify firewall status
ufw status verbose
ufw show added
ufw show added numbered

# Logging
ufw logging on                 # Enable firewall logging
ufw logging medium             # Log level (low, medium, high)

# View logs
tail -f /var/log/ufw.log
grep UFW /var/log/syslog | tail -20
```

### Advanced UFW Rules

```bash
#!/bin/bash
# advanced-firewall-rules.sh - Complex firewall scenarios

# ALLOW ONLY FROM SPECIFIC IPS (Whitelist approach)
ufw allow from 192.168.1.100 to any port 3306  # MySQL from admin PC
ufw allow from 10.0.0.0/8 to any port 5432     # PostgreSQL from subnet
ufw allow from 203.0.113.45 to any port 8443   # HTTPS from specific IP

# DENY SPECIFIC TRAFFIC
ufw deny from 192.168.1.50                      # Block entire IP
ufw deny from 192.168.0.0/24                    # Block entire subnet
ufw deny in on eth0 from 203.0.113.1            # Block on specific interface

# ALLOW SPECIFIC PORT RANGE
ufw allow 5000:6000/tcp                         # Range of ports
ufw allow from 10.0.0.0/8 to any port 5000:6000/tcp

# ALLOW TRAFFIC BETWEEN INTERFACES
ufw allow in on eth0 to 192.168.1.1 port 443    # Traffic to specific address
ufw allow out on eth1 to 192.168.2.0/24 port 53 # DNS queries out eth1

# LIMIT AND LOG
ufw limit 22/tcp log              # Rate limit and log SSH
ufw allow 443/tcp log             # Allow but log HTTPS
ufw deny 23/tcp log-all           # Deny with full logging

# MANAGE RULES
ufw show numbered                  # Show rules with numbers
ufw delete allow 80/tcp           # Delete rule by name
ufw delete 5                      # Delete rule by number
ufw insert 1 allow from 10.0.0.0/8  # Insert at position 1
ufw reload                        # Reload all rules

# RESET FIREWALL (dangerous!)
ufw reset                         # Clear all rules
```

## 30.3 Network Segmentation with VLANs

### VLAN Configuration

```bash
#!/bin/bash
# vlan-setup.sh - Configure network segmentation

# Create virtual VLAN interfaces
ip link add link eth0 name eth0.100 type vlan id 100  # Management
ip link add link eth0 name eth0.200 type vlan id 200  # Control Plane
ip link add link eth0 name eth0.300 type vlan id 300  # Workers
ip link add link eth0 name eth0.400 type vlan id 400  # Database
ip link add link eth0 name eth0.500 type vlan id 500  # User/App

# Bring up interfaces
ip link set eth0.100 up
ip link set eth0.200 up
ip link set eth0.300 up
ip link set eth0.400 up
ip link set eth0.500 up

# Assign IP addresses
ip addr add 10.100.0.1/24 dev eth0.100    # Management: 10.100.x.x
ip addr add 10.200.0.1/24 dev eth0.200    # Control Plane: 10.200.x.x
ip addr add 10.300.0.1/24 dev eth0.300    # Workers: 10.300.x.x
ip addr add 10.400.0.1/24 dev eth0.400    # Database: 10.400.x.x
ip addr add 10.500.0.1/24 dev eth0.500    # Apps: 10.500.x.x

# Configure inter-VLAN routing (if on same device)
sysctl -w net.ipv4.ip_forward=1

# Configure firewall rules between VLANs
# Block all inter-VLAN by default
ufw default deny in
ufw default deny out

# Allow specific VLAN-to-VLAN communication
ufw allow from 10.200.0.0/24 to 10.300.0.0/24 port 6443    # CP to workers
ufw allow from 10.300.0.0/24 to 10.400.0.0/24 port 5432    # Workers to DB
ufw allow from 10.500.0.0/24 to 10.300.0.0/24 port 8080    # Apps to workers

# Persistent VLAN configuration (Ubuntu)
# /etc/netplan/01-vlan.yaml
cat > /etc/netplan/01-vlan.yaml << 'NETPLAN'
network:
  version: 2
  ethernets:
    eth0:
      dhcp4: false
  vlans:
    vlan100:
      id: 100
      link: eth0
      addresses:
      - 10.100.0.1/24
    vlan200:
      id: 200
      link: eth0
      addresses:
      - 10.200.0.1/24
    vlan300:
      id: 300
      link: eth0
      addresses:
      - 10.300.0.1/24
    vlan400:
      id: 400
      link: eth0
      addresses:
      - 10.400.0.1/24
NETPLAN

# Apply configuration
netplan apply

# Verify
ip link show type vlan
ip addr show | grep 10.
```

### Network Segmentation Strategy

```
VLAN DESIGN FOR DEVSECOPS LAB:

VLAN 100: Management (10.100.0.0/24)
├─ Hypervisor management
├─ Out-of-band management
├─ Monitoring consoles
└─ Only authorized admins access

VLAN 200: Kubernetes Control Plane (10.200.0.0/24)
├─ 3 control plane nodes
├─ etcd cluster
├─ API server
└─ Only control plane traffic

VLAN 300: Kubernetes Workers (10.300.0.0/24)
├─ 2+ worker nodes
├─ Container workloads
├─ Pod traffic
└─ Pod-to-pod communication only

VLAN 400: Databases (10.400.0.0/24)
├─ PostgreSQL servers
├─ Redis cache
├─ Elasticsearch nodes
└─ No user access, only apps

VLAN 500: Applications/Users (10.500.0.0/24)
├─ User connections
├─ Application ingress
├─ Load balancers
└─ Public-facing services

TRAFFIC FLOWS (Allowed):
├─ Management → All (admin access)
├─ CP → Workers (orchestration)
├─ Workers ↔ Workers (pod-to-pod)
├─ Workers → Database (app queries)
├─ Apps → Workers (ingress)
└─ Internet → Apps (user traffic)

TRAFFIC FLOWS (Blocked):
├─ Workers → CP (no worker-initiated)
├─ Database → Workers (not db-initiated)
├─ Apps → Database (only through workers)
├─ User → Management (strict isolation)
└─ User → Database (no direct access)

BENEFITS:
✅ Lateral movement limited to same VLAN
✅ Compromised app cannot reach database directly
✅ Compromised worker cannot reach control plane
✅ Administrative traffic isolated
✅ Blast radius of breaches reduced
```

## 30.4 Intrusion Detection with Suricata

### Suricata Installation & Configuration

```bash
#!/bin/bash
# suricata-setup.sh - Deploy Suricata IDS/IPS

# Install Suricata
apt-get install -y suricata

# Install Suricata repository for latest rules
add-apt-repository ppa:oisf/suricata-stable
apt-get update
apt-get install -y suricata

# Main configuration
cat > /etc/suricata/suricata.yaml << 'SURICATA'
vars:
  address-groups:
    HOME_NET: "[10.0.0.0/8]"
    EXTERNAL_NET: "!$HOME_NET"
    HTTP_SERVERS: "[10.0.2.0/24]"
    SMTP_SERVERS: "[10.0.2.25]"
    DNS_SERVERS: "[10.0.2.10]"
    TELNET_SERVERS: "[]"
    AIM_SERVERS: "[]"

  port-groups:
    HTTP_PORTS: "[80, 311, 383, 591, 593, 901, 1311, 7001, 8000, 8008, 8028, 8080, 8085, 8090, 8118, 8123, 8180, 8244, 8888, 11371]"
    SHELLCODE_PORTS: "!80"
    ORACLE_PORTS: 1521
    SSH_PORTS: 22
    FTP_PORTS: 21
    DCERPC_PORTS: "[135, 139, 445]"
    NETBIOS_PORTS: "[137, 138, 139]"
    SNMP_PORTS: 161
    VXLAN_PORTS: 4789

default-rule-path: /etc/suricata/rules

rule-files:
  - suricata.rules
  - custom.rules

# Inline mode (IPS - active blocking)
runmode: workers
af-packet:
  - interface: eth0
    cluster-id: 1
    cluster-type: cluster_flow
  - interface: eth1
    cluster-id: 2
    cluster-type: cluster_flow

# Output formats
outputs:
  - eve-log:
      enabled: yes
      filename: eve.json
      types:
        - alert
        - http
        - dns
        - tls
        - flow
        - stats

  - file-data-log:
      enabled: yes

  - drop-log:
      enabled: yes
      filename: drop.log

# Threat detection
threat-detection:
  tls:
    enabled: yes
    ja3-fingerprinting: yes
  ssh:
    enabled: yes
  http:
    enabled: yes

# Performance tuning
threading:
  set-cpu-affinity: no
  cpu-affinity:
    - management-cpu-set:
        cpu: [0]
    - receive-cpu-set:
        cpu: [1]
    - decode-cpu-set:
        cpu: [2, 3]
    - stream-cpu-set:
        cpu: [4, 5]
    - detect-cpu-set:
        cpu: [6, 7]

        cpu: [8]

logging:
  default-log-level: notice
  outputs:
  - console:
      enabled: yes
  - file:
      enabled: yes
      filename: /var/log/suricata/suricata.log

SURICATA

# Download rule sets
cd /etc/suricata/rules

# Download ET Open ruleset
wget https://rules.emergingthreats.net/open/suricata/emerging.rules.tar.gz
tar xzf emerging.rules.tar.gz

# Start Suricata
systemctl enable suricata

# Monitor alerts in real-time
tail -f /var/log/suricata/eve.json | jq 'select(.event_type=="alert")'

# Generate statistics
suricatasc -c "stats" /var/run/suricata/suricata-command.socket

# Update rules regularly
0 3 * * * /usr/bin/suricata-update && systemctl reload suricata
```

### Suricata Rule Examples

```bash
# /etc/suricata/rules/custom.rules

# Alert on SQL injection attempts
alert http any any -> any any (msg:"ET SQL Injection - SELECT"; \
  content:"SELECT"; http_uri; nocase; content:"UNION"; http_uri; nocase; \
  sid:1000001; rev:1;)

# Alert on suspicious DNS queries
alert dns any any -> any 53 (msg:"ET DNS Query for Known Malware Domain"; \
  dns.query; content:".ru"; \
  sid:1000002; rev:1;)

# Alert on brute force attempts (high number of failed SSH)
alert ssh any any -> any 22 (msg:"SSH Brute Force Attempt"; \
  flow:to_server; byte_test:1,>,20,0,relative; \
  sid:1000003; rev:1;)

# Alert on buffer overflow attempts
alert tcp any any -> any any (msg:"Possible Buffer Overflow"; \
  content:!"|00|"; byte_test:4,>,0,0; \
  sid:1000004; rev:1;)

# Detect SSL/TLS certificate issues
alert tls any any -> any any (msg:"Suspicious SSL Certificate"; \
  tls.cert_subject; content:"*"; \
  sid:1000005; rev:1;)
```

## 30.5 DDoS Mitigation Strategies

### Rate Limiting & Throttling

```bash
#!/bin/bash
# ddos-mitigation.sh - Configure DDoS protection

# 1. RATE LIMITING IN IPTABLES
iptables -A INPUT -p tcp --dport 80 -m limit --limit 100/minute --limit-burst 200 -j ACCEPT
iptables -A INPUT -p tcp --dport 80 -j DROP

# 2. SYN COOKIE PROTECTION (kernel level)
echo "net.ipv4.tcp_syncookies = 1" >> /etc/sysctl.conf
sysctl -p

# 3. TCP BACKLOG CONFIGURATION
echo "net.ipv4.tcp_max_syn_backlog = 2048" >> /etc/sysctl.conf
echo "net.core.netdev_max_backlog = 5000" >> /etc/sysctl.conf

# 4. ABORT CONNECTION ON TIMEOUT
echo "net.ipv4.tcp_abort_on_overflow = 1" >> /etc/sysctl.conf

# 5. UFW RATE LIMITING
ufw limit 22/tcp                    # SSH rate limit
ufw limit 80/tcp                    # HTTP rate limit
ufw limit 443/tcp                   # HTTPS rate limit

# 6. FAIL2BAN FOR APPLICATION LEVEL
apt-get install -y fail2ban

cat > /etc/fail2ban/jail.local << 'FAIL2BAN'
[DEFAULT]
bantime = 3600
findtime = 600
maxretry = 5

[sshd]
enabled = true
port = ssh
logpath = /var/log/auth.log

[nginx-http-auth]
enabled = true
port = http,https
logpath = /var/log/nginx/error.log

[nginx-limit-req]
enabled = true
port = http,https
logpath = /var/log/nginx/error.log

[http-get-dos]
enabled = true
port = http,https
logpath = /var/log/nginx/access.log
maxretry = 100
findtime = 60

FAIL2BAN

systemctl enable fail2ban
systemctl restart fail2ban

# 7. HAProxy DDoS PROTECTION
# In HAProxy config:
cat >> /etc/haproxy/haproxy.cfg << 'HAPROXY'

frontend web
  bind *:80
  
  # Rate limiting: 100 requests per second per IP
  stick-table type ip size 100k expire 30s store http_req_rate(10s)
  http-request track-sc0 src
  http-request deny if { sc_http_req_rate(0) gt 100 }
  
  # Connection limits
  http-request deny if { src_conn_rate(abuse_connections) gt 10 }
  
  # Geo-blocking (if needed)
  http-request deny if { src_ip 203.0.113.0/24 }

HAPROXY

# 8. NGINX DDoS PROTECTION
cat > /etc/nginx/conf.d/ddos-protection.conf << 'NGINX'

# Limit connections per IP
limit_conn_zone $binary_remote_addr zone=addr:10m;
limit_conn addr 10;

# Limit requests per IP
limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s;
limit_req zone=api burst=20 nodelay;

# Disable slow attacks (limit request body size)
client_max_body_size 10m;
client_body_timeout 10s;

# Prevent slowloris (limit header timeout)
client_header_timeout 10s;

# Limit request method
if ($request_method !~ ^(GET|HEAD|POST|PUT|DELETE|OPTIONS)$) {
  return 405;
}

NGINX

systemctl reload nginx
```

### Advanced DDoS Detection

```bash
#!/bin/bash
# ddos-detection.sh - Monitor for DDoS attacks

# Monitor packet rate
watch -n 1 'netstat -an | grep ESTABLISHED | wc -l'

# Monitor connections per IP
netstat -an | grep ESTABLISHED | awk '{print $5}' | cut -d: -f1 | sort | uniq -c | sort -rn | head -20

# Monitor SYN flood
netstat -an | grep SYN | wc -l

# tcpdump to capture suspicious traffic
tcpdump -i eth0 -c 1000 'tcp[tcpflags] & tcp-syn != 0' | grep -oE '[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+' | sort | uniq -c | sort -rn

# Monitor with Suricata
tail -f /var/log/suricata/eve.json | jq 'select(.event_type=="flow" and .flow.pkts_toserver > 1000)'
```
